&НаСервере
Функция ЗаполнитьНужноеНаСервере(ОперацияСравнения, ТипДаты, ДатаОтправки)
    // 1. Подготовка запроса для отбора ведомостей
    Запрос = Новый Запрос;
    
    ДеньЧисло = День(ДатаОтправки);
    УсловиеИсключения = "";
    Если ДеньЧисло >= 1 И ДеньЧисло <= 10 Тогда
        УсловиеИсключения = "И НЕ (ДЕНЬ(Документ.Дата) = " + ДеньЧисло + ")";
    ИначеЕсли ДеньЧисло >= 11 И ДеньЧисло <= 20 Тогда
        УсловиеИсключения = "И НЕ (ДЕНЬ(Документ.Дата) % 2 = 0)";
    ИначеЕсли ДеньЧисло >= 21 И ДеньЧисло <= 31 Тогда
        УсловиеИсключения = "И НЕ (МЕСЯЦ(Документ.Дата) % 2 = 1)";
    КонецЕсли;
    
    Сравнение = ?(ОперацияСравнения = "Равно", "=", "<>");
    ПолеАгрегата = ?(ТипДаты = "Минимальная", "МинДатаДоговора", "МаксДатаДоговора");
    
    ТекстЗапроса = 
    "ВЫБРАТЬ
    |    Ведомость.Ссылка КАК Ведомость,
    |    Ведомость.Дата КАК ДатаДокумента,
    |    ВТ_Данные.МинДатаДоговора,
    |    ВТ_Данные.МаксДатаДоговора,
    |    ТЧ_ПервыйАктивный.НачалоДействия КАК ДатаПервогоАктивногоДоговора
    |ИЗ
    |    Документ.Ведомость КАК Ведомость
    |    ЛЕВОЕ СОЕДИНЕНИЕ (
    |        ВЫБРАТЬ
    |            ТЧ.Ссылка КАК Ведомость,
    |            МИНИМУМ(Договор.НачалоДействия) КАК МинДатаДоговора,
    |            МАКСИМУМ(Договор.НачалоДействия) КАК МаксДатаДоговора,
    |            МИНИМУМ(ВЫБОР КОГДА Договор.Активен = ИСТИНА ТОГДА ТЧ.НомерСтроки КОНЕЦ) КАК МинНомерАктивногоДоговора
    |        ИЗ
    |            Документ.Ведомость.Договоры КАК ТЧ
    |            ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник.Договоры КАК Договор ПО ТЧ.Договор = Договор.Ссылка
    |        СГРУППИРОВАТЬ ПО
    |            ТЧ.Ссылка
    |    ) КАК ВТ_Данные
    |        ПО Ведомость.Ссылка = ВТ_Данные.Ведомость
    |    ЛЕВОЕ СОЕДИНЕНИЕ Документ.Ведомость.Договоры КАК ТЧ_ПервыйАктивный
    |        ПО ВТ_Данные.Ведомость = ТЧ_ПервыйАктивный.Ссылка
    |        И ВТ_Данные.МинНомерАктивногоДоговора = ТЧ_ПервыйАктивный.НомерСтроки
    |ГДЕ
    |    ВТ_Данные.МинНомерАктивногоДоговора ЕСТЬ НЕ NULL
    |    И ТЧ_ПервыйАктивный.НачалоДействия " + Сравнение + " ВТ_Данные." + ПолеАгрегата + "
    |    " + УсловиеИсключения + "
    |
    |УПОРЯДОЧИТЬ ПО
    |    ДатаДокумента ВОЗР";
    
    Запрос.Текст = ТекстЗапроса;
    РезультатЗапроса = Запрос.Выполнить();
    Выборка = РезультатЗапроса.Выбрать();
    
    // 2. Собираем ссылки ведомостей для получения последних статусов
    СписокВедомостей = Новый СписокЗначений;
    Пока Выборка.Следующий() Цикл
        СписокВедомостей.Добавить(Выборка.Ведомость);
    КонецЦикла;
    
    // 3. Запрос последних статусов из регистра сведений
    ТаблицаСтатусов = Новый ТаблицаЗначений;
    ТаблицаСтатусов.Колонки.Добавить("Ведомость");
    ТаблицаСтатусов.Колонки.Добавить("Статус");
    
    Если СписокВедомостей.Количество() > 0 Тогда
        ЗапросСтатусов = Новый Запрос;
        ЗапросСтатусов.Текст = 
        "ВЫБРАТЬ
        |    История.Документ КАК Ведомость,
        |    История.Статус КАК Статус
        |ИЗ
        |    РегистрСведений.ИсторияСтатусовДокументов КАК История
        |ГДЕ
        |    История.Документ В (&СписокВедомостей)
        |    И История.ДатаЗаписи = (
        |        ВЫБРАТЬ МАКСИМУМ(ИсторияВ.ДатаЗаписи)
        |        ИЗ
        |            РегистрСведений.ИсторияСтатусовДокументов КАК ИсторияВ
        |        ГДЕ
        |            ИсторияВ.Документ = История.Документ
        |    )";
        
        ЗапросСтатусов.УстановитьПараметр("СписокВедомостей", СписокВедомостей);
        РезультатСтатусов = ЗапросСтатусов.Выполнить();
        ВыборкаСтатусов = РезультатСтатусов.Выбрать();
        
        Пока ВыборкаСтатусов.Следующий() Цикл
            СтрокаТаблицы = ТаблицаСтатусов.Добавить();
            СтрокаТаблицы.Ведомость = ВыборкаСтатусов.Ведомость;
            СтрокаТаблицы.Статус = ВыборкаСтатусов.Статус;
        КонецЦикла;
    КонецЕсли;
    
    // 4. Формируем итоговую таблицу результата
    ИтоговаяТаблица = Новый ТаблицаЗначений;
    ИтоговаяТаблица.Колонки.Добавить("Ведомость", Новый ОписаниеТипов("ДокументСсылка.Ведомость"));
    ИтоговаяТаблица.Колонки.Добавить("Статус", Новый ОписаниеТипов("Строка"));
    
    Выборка = РезультатЗапроса.Выбрать();
    Пока Выборка.Следующий() Цикл
        НоваяСтрока = ИтоговаяТаблица.Добавить();
        НоваяСтрока.Ведомость = Выборка.Ведомость;
        // Поиск статуса в ТаблицаСтатусов
        Найденный = ТаблицаСтатусов.Найти(Выборка.Ведомость, "Ведомость");
        Если Найденный <> Неопределено И Найденный.Статус = Перечисления.СтатусыДокументов.Акцептован Тогда
            НоваяСтрока.Статус = "Акцептован";
        Иначе
            НоваяСтрока.Статус = "";
        КонецЕсли;
    КонецЦикла;
    
    Возврат ИтоговаяТаблица;
КонецФункции

Процедура УдалитьЛишнее(Команда)
    Если Результат.Количество() = 0 Тогда
        Возврат;
    КонецЕсли;
    
    // Вызов серверного метода удаления строк с чётным количеством договоров
    РезультатУдаления = УдалитьЛишнееНаСервере(Результат);
    
    // Обновляем таблицу на форме
    Результат = РезультатУдаления.НоваяТаблица;
    
    // Вывод информации в зависимости от длины фамилии
    ДлинаФамилии = СтрДлина(ФамилияКандидата);
    ТекстСообщения = РезультатУдаления.ТекстСообщения;
    
    Если ДлинаФамилии < 6 Тогда
        Сообщить(ТекстСообщения);
    ИначеЕсли ДлинаФамилии >= 6 И ДлинаФамилии <= 7 Тогда
        СообщениеПользователю = Новый СообщениеПользователю;
        СообщениеПользователю.Текст = ТекстСообщения;
        СообщениеПользователю.Сообщить();
    ИначеЕсли ДлинаФамилии >= 8 И ДлинаФамилии <= 9 Тогда
        Форма = СоздатьПроизвольнуюФормуСообщения(ТекстСообщения);
        Форма.Открыть();
    ИначеЕсли ДлинаФамилии >= 10 И ДлинаФамилии <= 11 Тогда
        Элементы.ТекстСообщения.Заголовок = ТекстСообщения;
        Элементы.ТекстСообщения.Видимость = Истина;
    ИначеЕсли ДлинаФамилии >= 12 И ДлинаФамилии <= 13 Тогда
        ВызватьИсключение ТекстСообщения;
    Иначе
        ПоказатьПредупреждение(, ТекстСообщения);
    КонецЕсли;
КонецПроцедуры

&НаСервере
Функция УдалитьЛишнееНаСервере(ТаблицаРезультат)
    НоваяТаблица = ТаблицаРезультат.Скопировать();
    УдаленныеНомера = Новый Массив;
    
    Для каждого Строка Из НоваяТаблица Цикл
        // Получаем количество договоров в табличной части документа
        Запрос = Новый Запрос;
        Запрос.Текст = "ВЫБРАТЬ КОЛИЧЕСТВО(*) ИЗ Документ.Ведомость.Договоры ГДЕ Ссылка = &Ссылка";
        Запрос.УстановитьПараметр("Ссылка", Строка.Ведомость);
        КоличествоДоговоров = Запрос.Выполнить().Выбрать().Количество();
        
        Если КоличествоДоговоров % 2 = 0 Тогда
            УдаленныеНомера.Добавить(Строка.Ведомость.Номер);
        КонецЕсли;
    КонецЦикла;
    
    // Удаляем строки с чётным количеством
    Для Каждого НомерДок Из УдаленныеНомера Цикл
        НоваяТаблица.Удалить(НоваяТаблица.Найти(НомерДок, "Ведомость.Номер")); // упрощённо, лучше по ссылке
    КонецЦикла;
    
    // Сортируем номера по возрастанию как строки
    УдаленныеНомера.СортироватьПоЗначению();
    СписокНомеров = "";
    Для Каждого Номер Из УдаленныеНомера Цикл
        Если СписокНомеров <> "" Тогда СписокНомеров = СписокНомеров + ", "; КонецЕсли;
        СписокНомеров = СписокНомеров + Номер;
    КонецЦикла;
    
    ТекстСообщения = "Удалено документов: " + УдаленныеНомера.Количество() + " (" + СписокНомеров + ").";
    
    Результат = Новый Структура;
    Результат.Вставить("НоваяТаблица", НоваяТаблица);
    Результат.Вставить("ТекстСообщения", ТекстСообщения);
    
    Возврат Результат;
КонецФункции
